import { App } from '@octokit/app'
import { Octokit } from '@octokit/rest'
import { createAppAuth } from '@octokit/auth-app'

class GitHubAppAuth {

  private app: App
  private tokenCache: Map<number, {
    token: string
    expiresAt: Date
  }> = new Map()

  constructor() {
    const privateKey = process.env.GITHUB_APP_PRIVATE_KEY
    const appId = process.env.GITHUB_APP_ID

    if (!privateKey || !appId) {
      console.warn(
        '[GitHubAppAuth] GitHub App not configured — ' +
        'GITHUB_APP_ID / GITHUB_APP_PRIVATE_KEY env vars missing.'
      )
      return
    }

    const decodedKey = Buffer.from(privateKey, 'base64').toString('utf8')

    this.app = new App({
      appId: parseInt(appId),
      privateKey: decodedKey,
      webhooks: {
        secret: process.env.GITHUB_APP_WEBHOOK_SECRET ?? ''
      },
    })
  }

  isConfigured(): boolean
    return !!this.app

  async getInstallationOctokit(
    installationId: number
  ): Promise<Octokit>
    PURPOSE: Get an authenticated Octokit for a specific
    installation. Caches tokens until 5 minutes before expiry.

    IMPLEMENTATION:
    1. Check cache:
       const cached = this.tokenCache.get(installationId)
       const fiveMinutesFromNow = new Date(Date.now() + 5 * 60 * 1000)
       if (cached && cached.expiresAt > fiveMinutesFromNow) {
         return new Octokit({ auth: cached.token })
       }

    2. Request fresh token:
       const octokit = await this.app.getInstallationOctokit(
         installationId
       )
       Cache the token with expiry.
       Return octokit.

    3. If token request fails:
       throw new AppError(401,
         `Cannot get token for installation ${installationId}`,
         'GITHUB_APP_AUTH_FAILED'
       )

  async getAppOctokit(): Promise<Octokit>
    — Returns app-level (not installation-level) Octokit
    — Used for endpoints like /app/installations

  verifyWebhookSignature(
    payload: string | Buffer,
    signature: string
  ): boolean
    HMAC-SHA256 verification using GITHUB_APP_WEBHOOK_SECRET.
    Return false (do NOT throw) if signature invalid —
    the route handler decides what HTTP response to send.

    const sig = signature.startsWith('sha256=')
      ? signature : `sha256=${signature}`
    const expected = 'sha256=' + crypto
      .createHmac('sha256', process.env.GITHUB_APP_WEBHOOK_SECRET ?? '')
      .update(payload)
      .digest('hex')
    return crypto.timingSafeEqual(
      Buffer.from(sig),
      Buffer.from(expected)
    )

  getAppInstallUrl(): string
    return `https://github.com/apps/${
      process.env.GITHUB_APP_SLUG ?? 'cicd-reliability'
    }/installations/new`

}

export const githubAppAuth = new GitHubAppAuth()

import { githubAppAuth } from './app-auth'
import { commentFormatter } from './comment-formatter'

class CheckRunManager {

  async create(params: {
    installationId: number
    owner: string
    repo: string
    headSha: string
    name?: string
  }): Promise<number>
    — Create a check run in 'in_progress' status
    — name defaults to 'CI/CD Reliability Analysis'
    — Returns checkRunId

    const octokit = await githubAppAuth.getInstallationOctokit(
      params.installationId
    )
    const { data } = await octokit.rest.checks.create({
      owner: params.owner,
      repo: params.repo,
      name: params.name ?? 'CI/CD Reliability Analysis',
      head_sha: params.headSha,
      status: 'in_progress',
      started_at: new Date().toISOString(),
    })
    return data.id

  async complete(params: {
    installationId: number
    owner: string
    repo: string
    checkRunId: number
    grade: string
    score: number
    criticalCount: number
    highCount: number
    topFindings: TopFinding[]
    blockOnGrade: string | null
    dashboardUrl: string
    scanId: string
    repoId: string
  }): Promise<void>

    CONCLUSION LOGIC:
    1. Determine if this should block the PR:
       const gradeValue = { A: 5, B: 4, C: 3, D: 2, F: 1 }
       const shouldBlock = params.blockOnGrade !== null &&
         gradeValue[params.grade] <=
         gradeValue[params.blockOnGrade]

    2. Build conclusion:
       const conclusion = shouldBlock ? 'failure'
         : params.grade === 'A' || params.grade === 'B'
           ? 'success'
           : 'neutral'

    3. Build title:
       shouldBlock:
         "⛔ Pipeline blocked — Grade ${grade}, Score ${score}/100"
       grade A/B:
         "✅ Grade ${grade} — ${totalFindings} findings"
       else:
         "⚠️ Grade ${grade} — review recommended"

    4. Get summary from commentFormatter.formatCheckRunSummary()

    5. Update check run:
       await octokit.rest.checks.update({
         owner, repo, check_run_id: checkRunId,
         status: 'completed',
         completed_at: new Date().toISOString(),
         conclusion,
         output: { title, summary, text }
       })

  async fail(params: {
    installationId: number
    owner: string
    repo: string
    checkRunId: number
    error: string
  }): Promise<void>
    — Mark check run as failed with error message
    conclusion: 'cancelled'
    title: 'CI/CD Reliability Analysis — scan failed'
    summary: error message

}

class PRCommenter {

  async postOrUpdate(params: {
    installationId: number
    owner: string
    repo: string
    prNumber: number
    repoId: string
    commentBody: string
  }): Promise<number>
    — Post new comment or UPDATE existing one (never duplicate)

    IMPLEMENTATION:
    1. Get octokit for installation
    2. List all PR comments:
       const { data: comments } =
         await octokit.rest.issues.listComments({
           owner, repo, issue_number: prNumber, per_page: 100
         })
    3. Find our comment by marker:
       const existing = comments.find(c =>
         c.body?.includes(
           `<!-- cicd-reliability-scan-${repoId} -->`
         )
       )
    4a. If exists: UPDATE it
       await octokit.rest.issues.updateComment({
         owner, repo,
         comment_id: existing.id,
         body: commentBody
       })
       return existing.id
    4b. If not exists: CREATE it
       const { data } = await octokit.rest.issues.createComment({
         owner, repo,
         issue_number: prNumber,
         body: commentBody
       })
       return data.id

  async deleteOurComments(params: {
    installationId: number
    owner: string
    repo: string
    prNumber: number
    repoId: string
  }): Promise<void>
    — Used when installation is removed or repo is disconnected
    — Find all our comments and delete them

}

class PatchCommitter {

  async commitPatchDoc(params: {
    installationId: number
    owner: string
    repo: string
    branch: string
    scanId: string
    manualFixes: ManualFixInstruction[]
    appliedFixes: Array<{ ruleId: string, filePath: string }>
  }): Promise<{ commitUrl: string, commitSha: string }>

    PURPOSE: Commit a .md documentation file explaining all
    findings and fixes to .github/cicd-reliability/patches.md

    IMPLEMENTATION:
    1. Get octokit
    2. Build markdown content:
       const content = this.buildPatchDoc(params)
       — documentation markdown, placeholders are OK in .md files
         (they are instructional text, not code)

    3. Check if file exists to get SHA for update:
       let existingSha: string | undefined
       try {
         const existing = await octokit.rest.repos.getContent({
           owner, repo,
           path: '.github/cicd-reliability/patches.md',
           ref: branch
         })
         existingSha = (existing.data as { sha: string }).sha
       } catch { }

    4. Commit the file:
       const { data } =
         await octokit.rest.repos.createOrUpdateFileContents({
           owner, repo,
           path: '.github/cicd-reliability/patches.md',
           message: `docs: CI/CD reliability findings for scan
             ${params.scanId.slice(0, 8)}`,
           content: Buffer.from(content).toString('base64'),
           branch,
           sha: existingSha,
           committer: {
             name: 'CI/CD Reliability Bot',
             email: 'bot@cicd-reliability.io'
           }
         })
    5. Return { commitUrl: data.commit.html_url, commitSha: data.commit.sha }

  private buildPatchDoc(params): string
    — Build full markdown document
    — Sections: Overview, Auto-applied fixes, Manual review needed
    — For each manual fix: currentCode, guidance, referenceUrl
    — Clearly mark examples as examples, never as paste-ready code

}

        f.hasAutoFixes &&
        f.validationPassed &&
        f.fixedContent !== f.originalContent
      )

      if (safeFiles.length === 0 && params.manualFixes.length === 0) {
        return null  — nothing to do
      }

    STEP 2 — Create patch branch:
      const branchName = `cicd-reliability/fixes-${
        params.scanId.slice(0, 8)
      }`
      — Get baseBranch HEAD SHA:
      const { data: baseRef } = await octokit.rest.git.getRef({
        owner, repo,
        ref: `heads/${params.baseBranch}`
      })
      — Create branch:
      try {
        await octokit.rest.git.createRef({
          owner, repo,
          ref: `refs/heads/${branchName}`,
          sha: baseRef.object.sha
        })
      } catch (err) {
        if (err.status === 422) {
          — Branch already exists, add timestamp suffix
          branchName = `cicd-reliability/fixes-${
            params.scanId.slice(0, 8)
          }-${Date.now()}`
          — retry createRef with new name
        } else throw err
      }

    STEP 3 — Commit each safe fixed file:
      THE FINAL GUARD — runs for every single file:

      const skippedFiles: string[] = []
      const committedFiles: string[] = []

      for (const fileResult of safeFiles) {
        — MANDATORY: final placeholder check before any commit
        if (containsPlaceholder(fileResult.fixedContent)) {
          console.error(
            `[auto-pr-creator] CRITICAL: placeholder found in ` +
            `fixedContent for ${fileResult.filePath} — skipping. ` +
            `This should have been caught by an earlier guard.`
          )
          skippedFiles.push(fileResult.filePath)
          continue
        }

        — Get current file SHA for update:
        let currentFileSha: string | undefined
        try {
          const { data } = await octokit.rest.repos.getContent({
            owner, repo,
            path: fileResult.filePath,
            ref: branchName
          })
          currentFileSha = (data as { sha: string }).sha
        } catch { }

        — Commit the file:
        try {
          await octokit.rest.repos.createOrUpdateFileContents({
            owner, repo,
            path: fileResult.filePath,
            message: `fix(${fileResult.filePath}): apply CI/CD ` +
              `reliability fixes [${
                fileResult.appliedPatchRuleIds.join(', ')
              }]`,
            content: Buffer.from(
              fileResult.fixedContent
            ).toString('base64'),
            branch: branchName,
            sha: currentFileSha,
            committer: {
              name: 'CI/CD Reliability Bot',
              email: 'bot@cicd-reliability.io'
            }
          })
          committedFiles.push(fileResult.filePath)
        } catch (err) {
          console.error(
            `[auto-pr-creator] Failed to commit ${fileResult.filePath}:`,
            err.message
          )
          skippedFiles.push(fileResult.filePath)
        }
      }

    STEP 4 — Commit patch documentation:
      Always commit the patches.md regardless of whether files changed,
      so the PR has at least one commit even if all auto-fixes failed:
      await patchCommitter.commitPatchDoc({
        installationId: params.installationId,
        owner, repo,
        branch: branchName,
        scanId: params.scanId,
        manualFixes: params.manualFixes,
        appliedFixes: safeFiles.map(f => ({
          ruleId: f.appliedPatchRuleIds.join(', '),
          filePath: f.filePath
        }))
      })

    STEP 5 — Open the PR:
      const prBody = committedFiles.length > 0
        ? formatPRBody({ ...params, appliedFixes: ..., headBranch: branchName })
        : formatNoFixesPRBody({ ...params })

      const { data: pr } = await octokit.rest.pulls.create({
        owner, repo,
        title: committedFiles.length > 0
          ? `fix: CI/CD reliability improvements ` +
            `(${committedFiles.length} file${
              committedFiles.length > 1 ? 's' : ''
            } fixed)`
          : `docs: CI/CD reliability findings — manual review needed`,
        body: prBody,
        head: branchName,
        base: params.baseBranch,
        maintainer_can_modify: true
      })

      — Try to add labels (ignore if labels don't exist):
      try {
        await octokit.rest.issues.addLabels({
          owner, repo,
          issue_number: pr.number,
          labels: ['ci/cd', 'reliability', 'automated']
        })
      } catch { }

    STEP 6 — Return:
      return {
        prUrl: pr.html_url,
        prNumber: pr.number,
        headBranch: branchName,
        filesChanged: committedFiles.length,
        skippedFiles
      }

}

export async function processPushEvent(payload: {
  ref: string
  after: string
  before: string
  commits: Array<{
    id: string
    modified: string[]
    added: string[]
    removed: string[]
  }>
  repository: {
    id: number
    name: string
    full_name: string
    private: boolean
    default_branch: string
    owner: { login: string }
  }
  installation: { id: number }
  sender: { login: string }
}, deliveryId: string): Promise<void>

  STEP 1 — Check if this push touches CI/CD files:
    const cicdPatterns = [
      '.github/workflows/',
      '.gitlab-ci.yml',
      'Dockerfile',
      'docker-compose',
      'Jenkinsfile',
      'k8s/',
      'kubernetes/',
      'helm/',
    ]
    const changedFiles = payload.commits.flatMap(c => [
      ...c.modified, ...c.added
    ])
    const hasCicdChanges = changedFiles.some(f =>
      cicdPatterns.some(p => f.includes(p) || f === p)
    )

    If NOT hasCicdChanges:
      Log: "Push event skipped — no CI/CD files changed"
      Update event status = 'skipped'
      return

  STEP 2 — Ignore branch deletions:
    if (payload.after === '0000000000000000000000000000000000000000') {
      Update event status = 'skipped'
      return
    }

  STEP 3 — Check if repo is configured in github_app_repos:
    const appRepo = await db.select().from(githubAppRepos)
      .where(and(
        eq(githubAppRepos.installationId, payload.installation.id),
        eq(githubAppRepos.githubRepoId, payload.repository.id)
      ))

    If not found or !appRepo[0].autoScanEnabled:
      Update event status = 'skipped'
      return

  STEP 4 — Find or create platform repo record:
    let repoRecord = appRepo[0].repoId
      ? await db.select().from(repos).where(eq(repos.id, appRepo[0].repoId))
      : null

    If no repoRecord:
      — Auto-register the repo on first scan:
      [repoRecord] = await db.insert(repos).values({
        id: crypto.randomUUID(),
        repoUrl: `https://github.com/${payload.repository.full_name}`,
        name: payload.repository.full_name,
        provider: 'github',
        owner: payload.repository.owner.login,
        repoName: payload.repository.name,
        defaultBranch: payload.repository.default_branch,
        status: 'active',
        totalScans: 0,
        settings: { autoScanOnPush: true }
      }).returning()
      — Link back to github_app_repos:
      await db.update(githubAppRepos)
        .set({ repoId: repoRecord.id })
        .where(eq(githubAppRepos.id, appRepo[0].id))

  STEP 5 — Cache installation token in Redis:
    The ScanWorker fetches files using GitHub API. It needs a token.
    Store the installation token in Redis under the same key
    the scan worker already looks for:
    const octokit = await githubAppAuth.getInstallationOctokit(
      payload.installation.id
    )
    — Extract the installation token from octokit's auth:
    — Store in Redis: temp-token:{repoId} with 24hr TTL
    — This reuses the EXISTING mechanism scan.worker.ts already
      has for token lookup — zero changes to scan worker needed

  STEP 6 — Create check run (in_progress) if this push is on a branch
    associated with any open PRs:
    try {
      const prList = await octokit.rest.pulls.list({
        owner: payload.repository.owner.login,
        repo: payload.repository.name,
        state: 'open',
        head: `${payload.repository.owner.login}:${
          payload.ref.replace('refs/heads/', '')
        }`
      })
      if (prList.data.length > 0) {
        checkRunId = await checkRunManager.create({
          installationId: payload.installation.id,
          owner: payload.repository.owner.login,
          repo: payload.repository.name,
          headSha: payload.after
        })
      }
    } catch { }

  STEP 7 — Trigger scan via existing scan route logic:
    (DO NOT call the HTTP route — call the service directly)
    const scanId = crypto.randomUUID()
    await db.insert(scans).values({
      id: scanId,
      repoId: repoRecord.id,
      status: 'running',
      branch: payload.ref.replace('refs/heads/', ''),
      triggeredAt: new Date(),
      createdAt: new Date(),
      updatedAt: new Date(),
      totalFiles: 0, totalFindings: 0,
      criticalCount: 0, highCount: 0,
      mediumCount: 0, lowCount: 0,
    })

    const job = await scanQueue.add('scan-repo', {
      scanId,
      repoId: repoRecord.id,
      repoUrl: `https://github.com/${payload.repository.full_name}`,
      owner: payload.repository.owner.login,
      repoName: payload.repository.name,
      branch: payload.ref.replace('refs/heads/', ''),
      triggeredBy: 'github-app-push',
      installationId: payload.installation.id,
      headSha: payload.after,
      checkRunId,
    })

  STEP 8 — Register the onScanComplete callback:
    Store { scanId, installationId, owner, repo, checkRunId,
            prNumbers (from open PRs found above),
            appRepoConfig: appRepo[0] }
    in Redis as scan-app-context:{scanId} TTL 24hrs
    So the scan completion handler can retrieve this context.

  STEP 9 — Update github_app_events record:
    status = 'processing', scanId = scanId, checkRunId = checkRunId

### POST /webhooks/github-app

The GitHub webhook receiver.
CRITICAL: Must use express.raw() middleware to preserve the raw
body for HMAC signature verification.

USE express.raw({ type: '*/*' }) on this route ONLY.
The raw body is needed to verify the webhook signature.

HANDLER:

1. Read headers:
   const signature = req.headers['x-hub-signature-256'] as string
   const eventType = req.headers['x-github-event'] as string
   const deliveryId = req.headers['x-github-delivery'] as string

2. Verify signature using raw body:
   const rawBody = req.body as Buffer
   const valid = githubAppAuth.verifyWebhookSignature(rawBody, signature)
   If !valid: return 401 { error: 'Invalid signature' }

3. Idempotency check using deliveryId:
   const existing = await db.select().from(githubAppEvents)
     .where(eq(githubAppEvents.deliveryId, deliveryId))
   if (existing.length > 0):
     return 200 { message: 'Already processed' }

4. Parse payload:
   const payload = JSON.parse(rawBody.toString())

5. Insert event record immediately (status: 'received'):
   await db.insert(githubAppEvents).values({
     deliveryId,
     eventType,
     action: payload.action,
     installationId: payload.installation?.id,
     repoFullName: payload.repository?.full_name,
     senderLogin: payload.sender?.login,
   })

6. Return 202 IMMEDIATELY before processing:
   res.status(202).json({ message: 'Webhook received' })

7. Process event asynchronously (setImmediate):
   setImmediate(async () => {
     try {
       if (eventType === 'push') {
         await processPushEvent(payload, deliveryId)
       } else if (eventType === 'pull_request') {
         if (['opened','synchronize','reopened']
           .includes(payload.action)) {
           await processPREvent(payload, deliveryId)
         }
       } else if (eventType === 'installation' ||
                  eventType === 'installation_repositories') {
         await processInstallationEvent(payload)
       } else if (eventType === 'ping') {
         await db.update(githubAppEvents)
           .set({ status: 'completed', processedAt: new Date() })
           .where(eq(githubAppEvents.deliveryId, deliveryId))
       }
     } catch (err) {
       console.error('[webhook] Processing error:', err)
       await db.update(githubAppEvents)
         .set({ status: 'failed', errorMessage: err.message,
           processedAt: new Date() })
         .where(eq(githubAppEvents.deliveryId, deliveryId))
     }
   })
